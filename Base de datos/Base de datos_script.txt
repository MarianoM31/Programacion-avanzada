CREATE DATABASE AhorcadoDB;
GO

USE AhorcadoDB;
GO

-- Tabla de Palabras (Diccionario)
CREATE TABLE Palabras (
    id_palabra INT PRIMARY KEY IDENTITY(1,1),
    palabra NVARCHAR(20) NOT NULL UNIQUE CHECK (LEN(palabra) BETWEEN 5 AND 10)
);
GO

-- Tabla de Jugadores
CREATE TABLE Jugadores (
    id_jugador INT PRIMARY KEY,
    nombre NVARCHAR(100) NOT NULL
);
GO

-- Tabla de Niveles de Dificultad
CREATE TABLE Niveles (
    id_nivel INT PRIMARY KEY,
    nombre_nivel NVARCHAR(20) UNIQUE NOT NULL,
    tiempo_segundos INT NOT NULL
);
GO

INSERT INTO Niveles (id_nivel, nombre_nivel, tiempo_segundos)
VALUES

(2, 'Normal', 60),
(3, 'Difícil', 30);
GO

-- Tabla de Partidas
CREATE TABLE Partidas (
    id_partida INT PRIMARY KEY IDENTITY(1,1),
    id_jugador INT NOT NULL,
    id_nivel INT NOT NULL,
    id_palabra INT NOT NULL,
    fecha DATETIME NOT NULL DEFAULT GETDATE(),
    resultado VARCHAR(10) CHECK (resultado IN ('Ganada', 'Perdida')),
    FOREIGN KEY (id_jugador) REFERENCES Jugadores(id_jugador),
    FOREIGN KEY (id_nivel) REFERENCES Niveles(id_nivel),
    FOREIGN KEY (id_palabra) REFERENCES Palabras(id_palabra)
);
GO

-- Vista para escalafón de jugadores
CREATE VIEW Vista_Escalafon AS
SELECT 
    j.id_jugador,
    j.nombre,
    SUM(
        CASE 
            WHEN p.resultado = 'Ganada' AND n.nombre_nivel = 'Fácil' THEN 1
            WHEN p.resultado = 'Ganada' AND n.nombre_nivel = 'Normal' THEN 2
            WHEN p.resultado = 'Ganada' AND n.nombre_nivel = 'Difícil' THEN 3
            WHEN p.resultado = 'Perdida' AND n.nombre_nivel = 'Fácil' THEN -1
            WHEN p.resultado = 'Perdida' AND n.nombre_nivel = 'Normal' THEN -2
            WHEN p.resultado = 'Perdida' AND n.nombre_nivel = 'Difícil' THEN -3
            ELSE 0
        END
    ) AS Marcador,
    SUM(CASE WHEN p.resultado = 'Ganada' THEN 1 ELSE 0 END) AS Ganadas,
    SUM(CASE WHEN p.resultado = 'Perdida' THEN 1 ELSE 0 END) AS Perdidas
FROM Jugadores j
LEFT JOIN Partidas p ON j.id_jugador = p.id_jugador
LEFT JOIN Niveles n ON p.id_nivel = n.id_nivel
GROUP BY j.id_jugador, j.nombre;
GO
